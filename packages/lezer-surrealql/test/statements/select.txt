# Simple select all

SELECT * FROM test

==>

SurrealQL(SelectStatement(Keyword,Fields(Any),Keyword,Ident))

# Select specific fields

SELECT name, age FROM person

==>

SurrealQL(SelectStatement(Keyword,Fields(Predicate(Ident),Predicate(Ident)),Keyword,Ident))

# Select with alias

==>
