# Empty closure

|| { 1 }

==>

SurrealQL(Closure(Pipe,Pipe,Block(BraceOpen,Number(Int),BraceClose)))

# Closure with param

|$a| { $a }

==>

SurrealQL(Closure(Pipe,ParamDefinition(VariableName),Pipe,Block(BraceOpen,VariableName,BraceClose)))

# Closure with typed param

|$a: number| { $a + 1 }

==>

SurrealQL(Closure(Pipe,ParamDefinition(VariableName,Colon,Type(TypeName)),Pipe,Block(BraceOpen,BinaryExpression(VariableName,Operator,Number(Int)),BraceClose)))

# Closure with multiple params

|$a, $b| { $a + $b }

==>

SurrealQL(Closure(Pipe,ParamDefinition(VariableName),ParamDefinition(VariableName),Pipe,Block(BraceOpen,BinaryExpression(VariableName,Operator,VariableName),BraceClose)))

# Closure with return type

|$a: number| -> number { $a + 1 }

==>

SurrealQL(Closure(Pipe,ParamDefinition(VariableName,Colon,Type(TypeName)),Pipe,LookupRight,TypeName,Block(BraceOpen,BinaryExpression(VariableName,Operator,Number(Int)),BraceClose)))

# Closure with expression body

|$a| $a + 1

==>

SurrealQL(Closure(Pipe,ParamDefinition(VariableName),Pipe,BinaryExpression(VariableName,Operator,Number(Int))))

# Closure with expression body as argument

array::map($items, |$item| $item.name)

==>

SurrealQL(FunctionCall(FunctionName,ArgumentList(VariableName,Closure(Pipe,ParamDefinition(VariableName),Pipe,Path(VariableName,Subscript(Ident))))))

# Chained closures with expression bodies

[1, 2].map(|$x| $x * 2).filter(|$y| $y > 1)

==>

SurrealQL(Path(Array("[",Number(Int),Number(Int),"]"),Subscript(IdiomFunction(FunctionName,ArgumentList(Closure(Pipe,ParamDefinition(VariableName),Pipe,BinaryExpression(VariableName,Operator,Number(Int)))))),Subscript(IdiomFunction(FunctionName,ArgumentList(Closure(Pipe,ParamDefinition(VariableName),Pipe,BinaryExpression(VariableName,Operator(">"),Number(Int))))))))

# Closure returning a lookup

|| ->knows

==>

SurrealQL(Closure(Pipe,Pipe,Path(Lookup(LookupRight,Ident))))
