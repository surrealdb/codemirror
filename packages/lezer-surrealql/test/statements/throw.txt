# Throw string

THROW "An error occurred"

==>

SurrealQL(ThrowStatement(Keyword,String))

# Throw variable

THROW $error

==>

SurrealQL(ThrowStatement(Keyword,VariableName))

# Throw expression

THROW "Error: " + $msg

==>

SurrealQL(ThrowStatement(Keyword,BinaryExpression(String,Operator,VariableName)))

# Throw in where

SELECT * FROM t WHERE THROW 'no'

==>

SurrealQL(SelectStatement(Keyword,Fields(Any),Keyword,Ident,WhereClause(Keyword,ThrowStatement(Keyword,String))))
