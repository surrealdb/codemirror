# For loop with array

FOR $item IN [1, 2, 3] {
	CREATE item SET value = $item;
}

==>

SurrealQL(ForStatement(Keyword,VariableName,Keyword,Array("[",Number(Int),Number(Int),Number(Int),"]"),Block(BraceOpen,CreateStatement(Keyword,Ident,SetClause(Keyword,FieldAssignment(Ident,Operator,VariableName))),BraceClose)))

# For loop with variable

FOR $item IN $list {
	UPDATE item SET value = $item;
}

==>

SurrealQL(ForStatement(Keyword,VariableName,Keyword,VariableName,Block(BraceOpen,UpdateStatement(Keyword,Ident,SetClause(Keyword,FieldAssignment(Ident,Operator,VariableName))),BraceClose)))

# For loop with unparenthesized select

FOR $node IN SELECT * FROM node {
	LET $next = type::record("node", $node.id.id() + 1);
	RELATE $node->edge->$next SET read = rand::bool();
}

==>

SurrealQL(ForStatement(Keyword,VariableName,Keyword,SelectStatement(Keyword,Fields(Any),Keyword,Ident),Block(BraceOpen,LetStatement(Keyword,ParamDefinition(VariableName),FunctionCall(FunctionName,ArgumentList(String,BinaryExpression(Path(VariableName,Subscript(Ident),Subscript(IdiomFunction(FunctionName,ArgumentList))),Operator,Number(Int))))),RelateStatement(Keyword,VariableName,LookupRight,Ident,LookupRight,VariableName,SetClause(Keyword,FieldAssignment(Ident,Operator,FunctionCall(FunctionName,ArgumentList)))),BraceClose)))
